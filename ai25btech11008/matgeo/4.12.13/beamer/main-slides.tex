\begin{frame}{Question}
If the equation of the base of an equilateral triangle is 
\[
x+y=2, \quad \text{and the vertex is } (2,-1),
\]
find the length of the side of the triangle.
\end{frame}

\begin{frame}{Step 1: Vector Form}
Equation of base line:
\begin{align}
\vec{n}^T \vec{x} = 2 \tag{1}
\end{align}
where $\vec{n} = \begin{pmatrix}1 \\ 1\end{pmatrix}$.

Vertex:
\[
\vec{A} = \begin{pmatrix}2 \\ -1\end{pmatrix}.
\]
\end{frame}


\begin{frame}{Step 2: Foot of Perpendicular}
Distance from $\vec{A}$ to line:
\begin{align}
d = \frac{\vec{n}^T \vec{A} - 2}{\|\vec{n}\|}
= \frac{-1}{\sqrt{2}} \tag{2}
\end{align}

Foot of perpendicular:
\[
\vec{D} = \vec{A} - d \frac{\vec{n}}{\|\vec{n}\|}
= \begin{pmatrix}\tfrac{5}{2} \\ -\tfrac{1}{2}\end{pmatrix}.
\]
\end{frame}


\begin{frame}{Step 3: Altitude Relation}
In an equilateral triangle:
\begin{align}
AD = \frac{\sqrt{3}}{2} a \tag{3}
\end{align}

Here
\[
AD = \|\vec{A}-\vec{D}\|
= \frac{1}{\sqrt{2}}.
\]
\end{frame}


\begin{frame}{Step 4: Side Length}
Using (3):
\begin{align}
\frac{1}{\sqrt{2}} = \frac{\sqrt{3}}{2} a
\implies a = \sqrt{\tfrac{2}{3}} \tag{4}
\end{align}

\[
\boxed{a = \sqrt{\tfrac{2}{3}}}
\]
\end{frame}




\begin{frame}[fragile]{Python Code}
\begin{lstlisting}[language=Python]
import numpy as np
import matplotlib.pyplot as plt

# Given points
A = np.array([2, -1])     # Vertex
D = np.array([2.5, -0.5]) # Foot of perpendicular

# Side length of equilateral triangle
a = np.sqrt(2/3)

# Direction vector of the base (perpendicular to normal (1,1))
m = np.array([1, -1]) / np.sqrt(2)   # Unit vector along base

# Base vertices B and C
B = D + (a/2) * m
C = D - (a/2) * m


\end{lstlisting}
\end{frame}

\begin{lstlisting}[language=C]
#include <stdio.h>
#include <math.h>

int main() {
    double Ax=2, Ay=-1;
    double Dx=2.5, Dy=-0.5;

    double AD = sqrt((Ax-Dx)*(Ax-Dx) + (Ay-Dy)*(Ay-Dy));
    double a = (2 * AD) / sqrt(3);

    printf("Side length = %lf\n", a);
    return 0;
}
