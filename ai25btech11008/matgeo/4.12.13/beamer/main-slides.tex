\begin{frame}{Question}
If the equation of the base of an equilateral triangle is 
\[
x+y=2, \quad \text{and the vertex is } (2,-1),
\]
find the length of the side of the triangle and the coordinates of the other two vertices.
\end{frame}

\begin{frame}{Step 1: Vector Form}
Equation of base line:
\begin{align}
\vec{n}^T \vec{x} = 2 \tag{1}
\end{align}
where $\vec{n} = \begin{pmatrix}1 \\ 1\end{pmatrix}$.

Vertex:
\[
\vec{A} = \begin{pmatrix}2 \\ -1\end{pmatrix}.
\]
\end{frame}


\begin{frame}{Step 2: Foot of Perpendicular (Matrix Form)}
The foot $\vec{D}$ is obtained by:
\[
\vec{D} = \vec{A} - \frac{\vec{n}^T\vec{A}-2}{\vec{n}^T\vec{n}} \vec{n}.
\]

Substitute values:
\[
\vec{D} = \begin{pmatrix}2 \\ -1\end{pmatrix} - 
\frac{2-1-2}{2} \begin{pmatrix}1 \\ 1\end{pmatrix}
= \begin{pmatrix}\tfrac{5}{2} \\ -\tfrac{1}{2}\end{pmatrix}.
\]
\end{frame}


\begin{frame}{Step 3: Altitude Relation}
Altitude length:
\[
AD = \|\vec{A}-\vec{D}\| = \frac{1}{\sqrt{2}}.
\]

Equilateral triangle relation:
\[
AD = \frac{\sqrt{3}}{2}a \quad \implies \quad 
a = \sqrt{\tfrac{2}{3}}.
\]
\end{frame}


\begin{frame}{Step 4: Other Vertices}
Direction vector along base:
\[
\vec{m} = \begin{pmatrix}1 \\ -1\end{pmatrix}, \quad \|\vec{m}\|=\sqrt{2}.
\]

Vertices of base:
\[
\vec{B} = \vec{D} + \frac{a}{2}\frac{\vec{m}}{\|\vec{m}\|}, 
\quad 
\vec{C} = \vec{D} - \frac{a}{2}\frac{\vec{m}}{\|\vec{m}\|}.
\]

Final:
\[
B = \left(\tfrac{5}{2}+\tfrac{1}{2\sqrt{3}}, -\tfrac{1}{2}-\tfrac{1}{2\sqrt{3}}\right),
\quad
C = \left(\tfrac{5}{2}-\tfrac{1}{2\sqrt{3}}, -\tfrac{1}{2}+\tfrac{1}{2\sqrt{3}}\right).
\]
\end{frame}


\begin{frame}[fragile]{Python Code}
\begin{lstlisting}[language=Python]
import numpy as np
import matplotlib.pyplot as plt

A = np.array([2, -1])     # Vertex
n = np.array([1, 1])      # Normal to base
# Foot of perpendicular
D = A - ((n @ A - 2) / (n @ n)) * n

# Side length
a = np.sqrt(2/3)

# Base direction vector
m = np.array([1, -1]) / np.sqrt(2)

# Other vertices
B = D + (a/2)*m
C = D - (a/2)*m
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{C Code}
\begin{lstlisting}[language=C]
#include <stdio.h>
#include <math.h>

int main() {
    double Ax=2, Ay=-1;
    double Dx=2.5, Dy=-0.5;

    double AD = sqrt((Ax-Dx)*(Ax-Dx) + (Ay-Dy)*(Ay-Dy));
    double a = (2 * AD) / sqrt(3);

    printf("Side length = %lf\n", a);

    // Base direction vector normalized
    double mx=1/sqrt(2), my=-1/sqrt(2);

    double Bx = Dx + (a/2)*mx;
    double By = Dy + (a/2)*my;
    double Cx = Dx - (a/2)*mx;
    double Cy = Dy - (a/2)*my;

    printf("B = (%lf,%lf)\n", Bx, By);
    printf("C = (%lf,%lf)\n", Cx, Cy);

    return 0;
}
\end{lstlisting}
