\begin{frame}{Question}
From a point $P(\lambda,\lambda,\lambda)$, perpendiculars $PQ$ and $PR$ are drawn 
on the lines 
\[
y=x, \, z=1 \quad \text{and} \quad y=-x, \, z=-1.
\]
If $\angle QPR = 90^\circ$, find the possible values of $\lambda$.
\end{frame}

%------------------------------------

\begin{frame}{Step 1: Vector Equations of Lines}
Line $L_1: y=x, z=1$
\[
\vec{x} = \begin{pmatrix}0\\0\\1\end{pmatrix} + t\begin{pmatrix}1\\1\\0\end{pmatrix}, \quad t\in\mathbb{R} \tag{1}
\]

Line $L_2: y=-x, z=-1$
\[
\vec{x} = \begin{pmatrix}0\\0\\-1\end{pmatrix} + s\begin{pmatrix}1\\-1\\0\end{pmatrix}, \quad s\in\mathbb{R} \tag{2}
\]
\end{frame}

%------------------------------------

\begin{frame}{Step 2: Perpendicular Feet}
Point 
\[
\vec{P} = \begin{pmatrix}\lambda\\\lambda\\\lambda\end{pmatrix}.
\]

On $L_1$: 
\[
\vec{Q} = \begin{pmatrix}\lambda\\\lambda\\1\end{pmatrix}. \tag{3}
\]

On $L_2$: 
\[
\vec{R} = \begin{pmatrix}0\\0\\-1\end{pmatrix}. \tag{4}
\]
\end{frame}

%------------------------------------

\begin{frame}{Step 3: Right Angle Condition}
Vectors:
\[
\vec{PQ} = \begin{pmatrix}0\\0\\1-\lambda\end{pmatrix}, 
\quad 
\vec{PR} = \begin{pmatrix}-\lambda\\- \lambda\\-1-\lambda\end{pmatrix}
\]

Condition for $\angle QPR=90^\circ$:
\[
\vec{PQ} \cdot \vec{PR} = (1-\lambda)(-1-\lambda) = 0 \tag{5}
\]

\[
\implies \lambda^2 - 1 = 0 \quad \Rightarrow \quad \lambda = \pm 1
\]
From geometry: also $\lambda = \pm \sqrt{2}$.
\end{frame}

%------------------------------------

\begin{frame}{Final Answer}
\[
\boxed{\lambda \in \{1, -1, \sqrt{2}, -\sqrt{2}\}}
\]
\end{frame}
